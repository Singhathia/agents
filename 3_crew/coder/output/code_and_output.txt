import math

def calculate_series(terms=10000):
    """
    Calculate the sum of the Leibniz series: 1 - 1/3 + 1/5 - 1/7 + ...
    for a given number of terms, then multiply by 4.
    """
    total = 0.0
    for i in range(terms):
        denominator = 2 * i + 1
        term = 1 / denominator
        if i % 2 == 0:  # positive term for even index (i=0 -> 1)
            total += term
        else:
            total -= term
    return total * 4
